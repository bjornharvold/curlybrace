import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.MongoDatabase;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import org.bson.Document;

public class Main {
    public static void main(String[] args) {
        try (MongoClient client = MongoClients.create("<your connection string>")) {
            MongoDatabase db = client.getDatabase("shop");
            MongoCollection<Document> collection = db.getCollection("orders");

            List<Document> pipeline = Arrays.asList(
                new Document("$match", new Document("status", new Document("$in", Arrays.asList("A", "B")))
                        .append("total", new Document("$gte", 100))),
                new Document("$lookup", new Document("from", "customers")
                        .append("localField", "customerId")
                        .append("foreignField", "_id")
                        .append("as", "customer")),
                new Document("$group", new Document("_id", "$customer.region")
                        .append("revenue", new Document("$sum", "$total"))
                        .append("orders", new Document("$count", new Document()))),
                new Document("$facet", new Document("top", Arrays.asList(
                            new Document("$sort", new Document("revenue", -1)),
                            new Document("$limit", 5)
                        ))
                        .append("all", Arrays.asList(new Document("$count", "n"))))
            );

            List<Document> results = collection.aggregate(pipeline).into(new ArrayList<>());
            System.out.println(results);
        }
    }
}
