import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.MongoDatabase;
import java.util.ArrayList;
import java.util.List;
import org.bson.Document;

public class Main {
    public static void main(String[] args) {
        try (MongoClient client = MongoClients.create("<your connection string>")) {
            MongoDatabase db = client.getDatabase("shop");
            MongoCollection<Document> collection = db.getCollection("orders");

            Document filter = new Document("status", "A").append("qty", new Document("$gt", 10));
            Document projection = new Document("name", 1).append("qty", 1).append("_id", 0);
            Document sort = new Document("qty", -1).append("name", 1);

            List<Document> results = collection.find(filter)
                .projection(projection)
                .sort(sort)
                .skip(20)
                .limit(10)
                .into(new ArrayList<>());
            System.out.println(results);
        }
    }
}
