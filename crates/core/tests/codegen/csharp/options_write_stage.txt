using MongoDB.Bson;
using MongoDB.Driver;

var client = new MongoClient("<your connection string>");
var db = client.GetDatabase("shop");
var collection = db.GetCollection<BsonDocument>("orders");

// Note: this pipeline writes with $out or $merge, which runs on the primary, so the read preference is left out.
var pipeline = new[] {
    new BsonDocument {
        { "$match", new BsonDocument { { "status", "A" } } },
    },
    new BsonDocument { { "$out", "archive" } },
};

var options = new AggregateOptions
{
    Comment = "archive",
};

var results = collection.Aggregate<BsonDocument>(pipeline, options).ToList();
results.ForEach(Console.WriteLine);
