using MongoDB.Bson;
using MongoDB.Driver;

var client = new MongoClient("<your connection string>");
var db = client.GetDatabase("shop");
var collection = db.GetCollection<BsonDocument>("orders");

var filter = new BsonDocument { { "status", "A" } };
var sort = new BsonDocument { { "total", -1 } };

var options = new FindOptions
{
    Collation = new Collation("en", caseLevel: false, caseFirst: CollationCaseFirst.Upper, strength: CollationStrength.Secondary, numericOrdering: true, alternate: CollationAlternate.Shifted),
    Hint = "status_1_total_-1",
    AllowDiskUse = true,
    MaxTime = TimeSpan.FromMilliseconds(5000),
    Comment = "nightly \"report\" 'q' \\ $x ${y} #{z} %s {0} \u202eend",
};

var results = collection.WithReadPreference(ReadPreference.SecondaryPreferred).Find(filter, options)
    .Sort(sort)
    .Limit(20)
    .ToList();
results.ForEach(Console.WriteLine);
