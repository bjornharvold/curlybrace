require 'mongo'

client = Mongo::Client.new('<your connection string>')
db = client.use('shop').database
collection = db['orders']

pipeline = [{ '$match' => { 'status' => 'A' } }, { '$sort' => { 'total' => -1 } }]

results = collection.with(read: { mode: :secondary_preferred }).aggregate(
  pipeline,
  collation: {
    'locale' => 'en',
    'strength' => 2,
    'caseLevel' => false,
    'caseFirst' => 'upper',
    'numericOrdering' => true,
    'alternate' => 'shifted',
  },
  hint: 'status_1_total_-1',
  allow_disk_use: true,
  max_time_ms: 5000,
  comment: "nightly \"report\" 'q' \\ $x ${y} \#{z} %s {0} \u{202e}end"
).to_a
puts results.inspect
