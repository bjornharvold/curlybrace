require 'mongo'

client = Mongo::Client.new('<your connection string>')
db = client.use('shop').database
collection = db['orders']

filter = { 'status' => 'A', 'qty' => { '$gt' => 10 } }
projection = { 'name' => 1, 'qty' => 1, '_id' => 0 }
sort = { 'qty' => -1, 'name' => 1 }

results = collection.find(
  filter,
  projection: projection,
  sort: sort,
  skip: 20,
  limit: 10
).to_a
puts results.inspect
