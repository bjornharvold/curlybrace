require 'mongo'

client = Mongo::Client.new('<your connection string>')
db = client.use('shop').database
collection = db['orders']

pipeline = [
  {
    '$match' => {
      'empty' => BSON::Binary.new([''].pack('H*'), :generic),
      'emptyUuid' => BSON::Binary.new([''].pack('H*'), :uuid),
      'oldUuid' => BSON::Binary.new(['000102030405060708090a0b0c0d0e0f'].pack('H*'), :uuid_old),
      'md5' => BSON::Binary.new(['000102030405060708090a0b0c0d0e0f'].pack('H*'), :md5),
      'shortUuid' => BSON::Binary.new(['0102030405'].pack('H*'), :uuid),
      'uuid' => BSON::Binary.new(['000102030405060708090a0b0c0d0e0f'].pack('H*'), :uuid),
    },
  },
]

results = collection.aggregate(pipeline).to_a
puts results.inspect
