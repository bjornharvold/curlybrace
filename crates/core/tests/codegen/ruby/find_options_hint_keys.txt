require 'mongo'

client = Mongo::Client.new('<your connection string>')
db = client.use('shop').database
collection = db['orders']

filter = { 'status' => 'A' }
sort = { 'total' => -1 }

results = collection.with(read: { mode: :nearest }).find(
  filter,
  sort: sort,
  limit: 20,
  collation: { 'locale' => 'fr_CA' },
  hint: { 'status' => 1, 'total' => -1 },
  allow_disk_use: false
).to_a
puts results.inspect
