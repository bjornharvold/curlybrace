require 'mongo'

client = Mongo::Client.new('<your connection string>')
db = client.use('shop').database
collection = db['orders']

pipeline = [
  {
    '$match' => { 'status' => { '$in' => ['A', 'B'] }, 'total' => { '$gte' => 100 } },
  },
  {
    '$lookup' => {
      'from' => 'customers',
      'localField' => 'customerId',
      'foreignField' => '_id',
      'as' => 'customer',
    },
  },
  {
    '$group' => {
      '_id' => '$customer.region',
      'revenue' => { '$sum' => '$total' },
      'orders' => { '$count' => {} },
    },
  },
  {
    '$facet' => {
      'top' => [{ '$sort' => { 'revenue' => -1 } }, { '$limit' => 5 }],
      'all' => [{ '$count' => 'n' }],
    },
  },
]

results = collection.aggregate(pipeline).to_a
puts results.inspect
