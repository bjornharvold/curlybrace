<?php

require 'vendor/autoload.php';

$client = new MongoDB\Client('<your connection string>');
$db = $client->selectDatabase('shop');
$collection = $db->selectCollection('orders');

$pipeline = [
    [
        '$match' => [
            'negativeZero' => -0.0,
            'int64Min' => new MongoDB\BSON\Int64('-9223372036854775808'),
            'int64Max' => new MongoDB\BSON\Int64('9223372036854775807'),
            'int32Min' => -2147483648,
            'int32Max' => 2147483647,
            'timestamp' => new MongoDB\BSON\Timestamp(4000000000, 3000000000),
        ],
    ],
];

$results = $collection->aggregate($pipeline)->toArray();
print_r($results);
