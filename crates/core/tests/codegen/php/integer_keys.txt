<?php

require 'vendor/autoload.php';

$client = new MongoDB\Client('<your connection string>');
$db = $client->selectDatabase('shop');
$collection = $db->selectCollection('orders');

$pipeline = [
    [
        '$match' => [
            'plain' => (object) ['b' => 1, '2' => 2, '1' => 3],
            'inOrder' => (object) ['1' => 1, '2' => 2, 'b' => 3],
            'leadingZero' => ['01' => 1, 'a' => 2],
            'nested' => [['x' => (object) ['z' => 1, '10' => 2, '9' => 3]]],
        ],
    ],
];

$results = $collection->aggregate($pipeline)->toArray();
print_r($results);
