filter := bson.D{{Key: "status", Value: "A"}}
sort := bson.D{{Key: "total", Value: int32(-1)}}

opts := options.Find().
	SetSort(sort).
	SetLimit(20).
	SetCollation(&options.Collation{Locale: "fr_CA"}).
	SetHint(bson.D{
		{Key: "status", Value: int32(1)},
		{Key: "total", Value: int32(-1)},
	}).
	SetAllowDiskUse(false)

cursor, err := collection.Clone(options.Collection().SetReadPreference(readpref.Nearest())).Find(context.TODO(), filter, opts)
if err != nil {
	log.Fatal(err)
}
var results []bson.M
if err := cursor.All(context.TODO(), &results); err != nil {
	log.Fatal(err)
}
