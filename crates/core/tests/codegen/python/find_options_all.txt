from pymongo import MongoClient, ReadPreference

client = MongoClient("<your connection string>")
db = client["shop"]
collection = db["orders"]

filter = {"status": "A"}
sort = [("total", -1)]

results = list(
    collection.with_options(read_preference=ReadPreference.SECONDARY_PREFERRED).find(
        filter,
        sort=sort,
        limit=20,
        collation={
            "locale": "en",
            "strength": 2,
            "caseLevel": False,
            "caseFirst": "upper",
            "numericOrdering": True,
            "alternate": "shifted",
        },
        hint="status_1_total_-1",
        allow_disk_use=True,
        max_time_ms=5000,
        comment="nightly \"report\" 'q' \\ $x ${y} #{z} %s {0} \u202eend",
    )
)
print(results)
