pipeline = [{"$match": {"status": "A"}}, {"$sort": {"total": -1}}]

results = list(
    collection.with_options(read_preference=ReadPreference.NEAREST).aggregate(
        pipeline,
        collation={"locale": "fr_CA"},
        hint=[("status", 1), ("total", -1)],
        allowDiskUse=False,
    )
)
print(results)
