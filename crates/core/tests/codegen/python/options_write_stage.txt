from pymongo import MongoClient

client = MongoClient("<your connection string>")
db = client["shop"]
collection = db["orders"]

# Note: this pipeline writes with $out or $merge, which runs on the primary, so the read preference is left out.
pipeline = [{"$match": {"status": "A"}}, {"$out": "archive"}]

results = list(
    collection.aggregate(
        pipeline,
        comment="archive",
    )
)
print(results)
