const pipeline = [
  { $match: { status: { $in: ["A", "B"] }, total: { $gte: 100 } } },
  {
    $lookup: {
      from: "customers",
      localField: "customerId",
      foreignField: "_id",
      as: "customer",
    },
  },
  {
    $group: {
      _id: "$customer.region",
      revenue: { $sum: "$total" },
      orders: { $count: {} },
    },
  },
  {
    $facet: {
      top: [{ $sort: { revenue: -1 } }, { $limit: 5 }],
      all: [{ $count: "n" }],
    },
  },
];

const results = await collection.aggregate(pipeline).toArray();
console.log(results);
