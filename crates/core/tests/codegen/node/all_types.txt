const {
  BSONRegExp,
  BSONSymbol,
  Binary,
  Code,
  DBRef,
  Decimal128,
  Double,
  Long,
  MaxKey,
  MinKey,
  MongoClient,
  ObjectId,
  Timestamp,
  UUID,
} = require("mongodb");

async function main() {
  const client = new MongoClient("<your connection string>");
  try {
    await client.connect();
    const db = client.db("shop");
    const collection = db.collection("orders");

    // Note: undefined/symbol/DBPointer values cannot be written in this language and were converted.
    const pipeline = [
      {
        $match: {
          double: 1.5,
          wholeDouble: new Double(1.0),
          nan: NaN,
          posInf: Infinity,
          negInf: -Infinity,
          string: "text",
          int32: 7,
          int64: Long.fromString("1099511627776"),
          bool: true,
          null: null,
          undefined: undefined,
          objectId: new ObjectId("65a1b2c3d4e5f60718293a4b"),
          date: new Date("2024-02-29T12:34:56.789Z"),
          dateBefore1970: new Date("1969-12-31T00:00:00.000Z"),
          dateYear10000: new Date(253402300800000),
          decimal: Decimal128.fromString("1.5E+3"),
          binary: new Binary(Buffer.from("AAEC/w==", "base64"), 0),
          uuid: new UUID("00010203-0405-0607-0809-0a0b0c0d0e0f"),
          userBinary: new Binary(Buffer.from("CQ==", "base64"), 128),
          regex: new BSONRegExp("^a.*z$", "i"),
          timestamp: new Timestamp({ t: 1700000000, i: 3 }),
          minKey: new MinKey(),
          maxKey: new MaxKey(),
          code: new Code("function () { return 1; }"),
          codeWithScope: new Code("x + 1", { x: 41 }),
          symbol: new BSONSymbol("sym"),
          dbPointer: new DBRef("shop.orders", new ObjectId("65a1b2c3d4e5f60718293a4b")),
          emptyDoc: {},
          emptyArray: [],
          nested: { a: { b: [1, [2, [3]], { c: "d" }] } },
          "needs quoting": 1,
          "dotted.key": 2,
          $dollar: 3,
          ["__proto__"]: 4,
          "": 5,
        },
      },
    ];

    const results = await collection.aggregate(pipeline).toArray();
    console.log(results);
  } finally {
    await client.close();
  }
}

main().catch(console.error);
