const collection = db.getSiblingDB("shop").getCollection("orders");

const filter = { status: "A" };
const sort = { total: -1 };

const results = collection
  .find(filter)
  .sort(sort)
  .limit(20)
  .collation({
    locale: "en",
    strength: 2,
    caseLevel: false,
    caseFirst: "upper",
    numericOrdering: true,
    alternate: "shifted",
  })
  .hint("status_1_total_-1")
  .allowDiskUse(true)
  .maxTimeMS(5000)
  .comment("nightly \"report\" 'q' \\ $x ${y} #{z} %s {0} \u202eend")
  .readPref("secondaryPreferred")
  .toArray();
printjson(results);
