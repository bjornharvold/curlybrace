const collection = db.getSiblingDB("shop").getCollection("orders");

const filter = { status: "A", qty: { $gt: 10 } };
const projection = { name: 1, qty: 1, _id: 0 };
const sort = { qty: -1, name: 1 };

const results = collection
  .find(filter, projection)
  .sort(sort)
  .skip(20)
  .limit(10)
  .toArray();
printjson(results);
