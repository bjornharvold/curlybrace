use futures::TryStreamExt;
use mongodb::Client;
use mongodb::bson::{Document, doc};
use mongodb::options::{
    Collation, CollationAlternate, CollationCaseFirst, CollationStrength, Hint, ReadPreference,
    SelectionCriteria,
};
use std::time::Duration;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let client = Client::with_uri_str("<your connection string>").await?;
    let db = client.database("shop");
    let collection = db.collection::<Document>("orders");

    let pipeline = vec![
        doc! { "$match": { "status": "A" } },
        doc! { "$sort": { "total": -1_i32 } },
    ];

    let mut cursor = collection
        .aggregate(pipeline)
        .collation(
            Collation::builder()
                .locale("en")
                .strength(CollationStrength::Secondary)
                .case_level(false)
                .case_first(CollationCaseFirst::Upper)
                .numeric_ordering(true)
                .alternate(CollationAlternate::Shifted)
                .build(),
        )
        .hint(Hint::Name("status_1_total_-1".to_string()))
        .allow_disk_use(true)
        .max_time(Duration::from_millis(5000))
        .comment("nightly \"report\" 'q' \\ $x ${y} #{z} %s {0} \u{202e}end")
        .selection_criteria(SelectionCriteria::ReadPreference(
            ReadPreference::SecondaryPreferred { options: None },
        ))
        .await?;
    while let Some(item) = cursor.try_next().await? {
        println!("{item}");
    }

    Ok(())
}
